## SoundService Requirements
```dart
class SoundService extends GetxService {
  // Background music: use FlameAudio.bgm
  Future<void> playBgMusic() async => FlameAudio.bgm.play('bg_music.mp3', volume: 0.4);
  Future<void> stopBgMusic() async => FlameAudio.bgm.stop();
  Future<void> pauseBgMusic() async => FlameAudio.bgm.pause();
  Future<void> resumeBgMusic() async => FlameAudio.bgm.resume();

  // Sound effects: use FlameAudio.play
  Future<void> playEatApple() async => FlameAudio.play('eat_apple.mp3', volume: 0.8);
  Future<void> playLevelComplete() async => FlameAudio.play('level_complete.mp3');
  Future<void> playGameOver() async => FlameAudio.play('game_over.mp3');
  Future<void> playButtonTap() async => FlameAudio.play('button_tap.mp3', volume: 0.5);
}
```
